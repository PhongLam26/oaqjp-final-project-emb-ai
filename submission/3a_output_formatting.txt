"""Call IBM Watson NLP to score the five supported emotions."""

import requests

WATSON_URL = (
    "https://sn-watson-emotion.labs.skills.network/v1/"
    "watson.runtime.nlp.v1/NlpService/EmotionPredict"
)
MODEL_ID = "emotion_aggregated-workflow_lang_en_stock"
EMOTION_NAMES = ("anger", "disgust", "fear", "joy", "sadness")
REQUEST_TIMEOUT = 30


def _error_result(message, status_code):
    """Return a consistent six-field result with an explicit error."""
    result = {name: None for name in EMOTION_NAMES}
    result["dominant_emotion"] = None
    result["error"] = {"status_code": status_code, "message": message}
    return result


def _empty_scores():
    """Return the six rubric fields with no scores."""
    result = {name: None for name in EMOTION_NAMES}
    result["dominant_emotion"] = None
    return result


def emotion_detector(text_to_analyze):
    """Return Watson emotion scores and the dominant emotion for text."""
    if not isinstance(text_to_analyze, str) or not text_to_analyze.strip():
        return _empty_scores()

    headers = {
        "Accept": "application/json",
        "grpc-metadata-mm-model-id": MODEL_ID,
    }
    payload = {"raw_document": {"text": text_to_analyze}}

    try:
        response = requests.post(
            WATSON_URL,
            json=payload,
            headers=headers,
            timeout=REQUEST_TIMEOUT,
        )
    except requests.RequestException:
        return _error_result(
            "The Watson NLP service could not be reached. Please try again later.",
            502,
        )

    if response.status_code == 400:
        return _empty_scores()
    if response.status_code != 200:
        return _error_result(
            f"The Watson NLP service returned HTTP {response.status_code}.",
            502,
        )

    try:
        predictions = response.json()["emotionPredictions"]
        emotions = predictions[0]["emotion"]
        scores = {name: float(emotions[name]) for name in EMOTION_NAMES}
    except (KeyError, IndexError, TypeError, ValueError):
        return _error_result(
            "The Watson NLP service returned an invalid response.",
            502,
        )

    scores["dominant_emotion"] = max(EMOTION_NAMES, key=scores.get)
    return scores
